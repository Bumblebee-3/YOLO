cmake_minimum_required(VERSION 3.16)
project(TahoeBattery LANGUAGES CXX)

find_package(Qt6 REQUIRED COMPONENTS Qml)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "${QT6_INSTALL_PREFIX}" CACHE PATH "Install prefix" FORCE)
endif()
set(module_dir "${QT6_INSTALL_QML}/TahoeBattery")
add_library(tahoebattery MODULE plugin.cpp powersupply.cpp)
install(TARGETS tahoebattery LIBRARY DESTINATION "${module_dir}")
install(FILES qmldir DESTINATION "${module_dir}")

include(CTest)
if(BUILD_TESTING)
    add_executable(power-test power-test.cpp powersupply.cpp)
    target_link_libraries(power-test PRIVATE Qt6::Core)
    add_test(NAME power-supplies COMMAND power-test)
endif()
target_link_libraries(tahoebattery PRIVATE Qt6::Qml)
